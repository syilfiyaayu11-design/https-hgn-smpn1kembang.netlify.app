Integrasi Database Google Spreadsheets

Aplikasi ini menyimpan perubahan secara otomatis di memori perangkat dan dapat disinkronkan secara langsung ke Google Spreadsheets menggunakan Google Apps Script Web App.

Kosongkan jika ingin menjalankan mode penyimpanan lokal otomatis (tetap tersimpan saat halaman ditutup).

Kode Google Apps Script (Untuk Spreadsheets Anda) 
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  
  // 1. Sheet Data Guru
  var sheetGuru = ss.getSheetByName("Data_Guru") || ss.insertSheet("Data_Guru");
  sheetGuru.clear();
  sheetGuru.appendRow(["ID", "Nama Guru", "Mata Pelajaran", "NIP", "Status Kunci"]);
  data.teachers.forEach(function(t) {
    sheetGuru.appendRow([t.id, t.name, t.subject, t.nip || "-", data.isLocked ? "TERKUNCI" : "AKTIF"]);
  });

  // 2. Sheet Kategori
  var sheetKat = ss.getSheetByName("Kategori") || ss.insertSheet("Kategori");
  sheetKat.clear();
  sheetKat.appendRow(["ID", "Nama Kategori", "Deskripsi"]);
  data.categories.forEach(function(c) {
    sheetKat.appendRow([c.id, c.title, c.description]);
  });

  // 3. Sheet Hasil Vote
  var sheetVote = ss.getSheetByName("Hasil_Vote") || ss.insertSheet("Hasil_Vote");
  sheetVote.clear();
  sheetVote.appendRow(["Waktu", "Nama Murid", "Kelas", "Detail Pilihan (JSON)"]);
  data.votes.forEach(function(v) {
    sheetVote.appendRow([v.timestamp, v.studentName, v.studentClass, JSON.stringify(v.choices)]);
  });

  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  // Mengembalikan status koneksi aktif
  return ContentService.createTextOutput(JSON.stringify({status: "connected", school: "SMPN 1 Kembang"})).setMimeType(ContentService.MimeType.JSON);
}
